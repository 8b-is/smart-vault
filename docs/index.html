Quick Start & M1 Global Install

# Install from source cleanly
cargo install --git https://github.com/8b-is/smart-vault

# Preview and apply global M1 setup (~/.local/bin/smart-vault + sv symlink, signed via codesign + .sha256,
# Antigravity IDE/CLI ~/.gemini MCP + hooks, and skill linking)
smart-vault setup-global
smart-vault setup-global --apply

# Verify binary signature & SHA-256 in constant time
smart-vault verify-binary ~/.local/bin/smart-vault --sha256 "$(awk '{print $1}' ~/.local/bin/smart-vault.sha256)"
